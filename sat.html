//BLOCK SUBMISSION // npm install bcoin const bcoin = require('bcoin'); const { SPVNode } = bcoin; const { Block } = bcoin.primitives; async function startLightNode(network = 'testnet') { const node = new SPVNode({ network, // 'testnet' -> 'main' to jump prefix: `./spv-${network}`, memory: false, listen: false, // outbound only, good for cheap internet maxOutbound: 8, // SPV: only headers, no full chain }); await node.open(); await node.connect(); console.log(`SPV connected to ${network}, peers:`, node.pool.peers.size()); console.log(`headers synced:`, node.chain.height); return node; } async function submitBlock(node, blockHex) { const block = Block.fromRaw(Buffer.from(blockHex, 'hex')); // Local check using real bcoin validator, not my code // SPVNode will check header + PoW against its header chain console.log('block hash:', block.hash().toString('hex')); console.log('nonce:', block.nonce); // This is the real test: will the node's chain accept it? // If PoW is invalid, it will be rejected here and NOT broadcast node.pool.broadcastBlock(block); console.log('broadcast attempted via SPV pool'); return block; } // const node = await startLightNode('testnet'); // await submitBlock(node, 'your testnet block hex');
